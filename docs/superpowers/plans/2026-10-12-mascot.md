# Mini Justine Mascot Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A pixel Mini Justine that waves and greets in the hero, follows the pointer with its eyes, hops and talks when clicked, reacts to the theme toggle, and makes typing and envelope cameos in Tech stack and Contact.

**Architecture:** Pose data is text pixel maps in a pure module (`src/lib/mascot/sprites.js`) turned into one SVG path per colour, with the eyes on their own layers; copy lives in `src/lib/mascot/lines.js`. A server-safe `Sprite` draws a pose; `HeroMascot` and `Cameo` are small client islands. All idle motion is CSS in `globals.css`; reduced motion turns it off without changing markup.

**Tech Stack:** Next.js 16 App Router, React 19, Tailwind CSS v4, node:test.

**Spec:** `docs/superpowers/specs/2026-10-12-mascot-design.md`

## Global Constraints

- Branch `feat/mascot`. Stage explicit paths only (`git add -A`, `git add .`, `git add --all` are denied by the guard). Read `git diff --cached --name-only` before every commit, and run `git branch --show-current` in the same command as the commit.
- Edit existing files with the Edit tool (Write for new files or full replacements this plan gives); no scripted replaces.
- Copy is approved verbatim; no em-dashes in user-facing text.
- The mascot palette is character art and lives only in `src/lib/mascot/sprites.js`; the outline colour is the `--mascot-outline` token in `globals.css`. Every other colour on the page stays a token class.
- Reduced motion never changes markup (hydration): CSS turns animations off, and JS checks `matchMedia` only inside effects and event handlers. Mount-only values use `useSyncExternalStore` with a `null` server snapshot.
- No synchronous `setState` inside `useEffect` bodies (callbacks such as a `MutationObserver` or a timer may set state).
- Keep template literals single-level and single-line (house style).
- Every commit message ends with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Checks: `npm run lint && npm test` while working; the full suite `npm run lint && npm test && npm run check:hooks && npm run build` is claimed green on the pushed tip at the end.
- A dev server may be running on port 3000; don't run `npm run build` while it runs (shared `.next`). The Chrome extension tab pauses `requestAnimationFrame`: verify motion by classes, attributes and computed styles; screenshots after a 2.5 s wait.

import {
  BASE,
  EYE_ROWS,
  EYE_WHITES,
  PALETTE,
  POSES,
  POSE_LAYERS,
  PUPILS,
  SIZE,
  hasEyes,
  patch,
  spriteLayers,

// Turns "M{x} {y}h{n}v1h-{n}z" segments back into "x,y" cells.
const cells = (d) => {
  const out = [];
  for (const [, x, y, n] of d.matchAll(/M(\d+) (\d+)h(\d+)v1h-\d+z/g)) {
    for (let i = 0; i < Number(n); i += 1) out.push(`${Number(x) + i},${y}`);
  }
  return out;
};

test("every pose is SIZE x SIZE and uses only palette keys", () => {
  const allowed = new Set([".", ...Object.keys(PALETTE)]);
  for (const [name, map] of Object.entries(POSES)) {
    assert.equal(map.length, SIZE, `${name}: rows`);
    for (const row of map) {
      assert.equal(row.length, SIZE, `${name}: row width`);
      for (const ch of row) assert.ok(allowed.has(ch), `${name}: unknown pixel "${ch}"`);
    }
  }
});

test("patch writes text, clears with '.', and keeps pixels under ' '", () => {
  assert.deepEqual(patch(["....", "...."], [[0, 1, "AB"], [1, 0, "C D"]]), [".AB.", "C.D."]);
  assert.deepEqual(patch(["XXXX"], [[0, 1, "."]]), ["X.XX"]);
});

test("the eyes are where the eye layers expect them, except under sunglasses", () => {
  for (const name of ["idle", "waveA", "waveB", "typingA", "typingB", "envelope"]) {
    assert.ok(hasEyes(POSES[name]), `${name} should show eyes`);
  }
  assert.equal(hasEyes(POSES.sunglasses), false);
});

test("the outline sits only on empty pixels next to filled ones", () => {
  assert.deepEqual(withOutline(["S"]), [".O.", "OSO", ".O."]);
  assert.deepEqual(withOutline(["S."]), [".O..", "OSO.", ".O.."]);
});

test("the layers draw every filled and outline pixel exactly once", () => {
  for (const [name, map] of Object.entries(POSES)) {
    const expected = new Set();
    withOutline(map).forEach((row, y) =>
      [...row].forEach((ch, x) => {
        if (ch !== ".") expected.add(`${x},${y}`);
      }),
    );
    const { size, paths, whites, pupils } = spriteLayers(map);
    assert.equal(size, SIZE + 2);
    const drawn = paths.flatMap(({ d }) => cells(d));
    assert.equal(drawn.length, new Set(drawn).size, `${name}: a pixel is drawn twice`);
    assert.deepEqual(new Set(drawn), expected, `${name}: body covers exactly the outlined sprite`);
    if (hasEyes(map)) {
      // White sits under the pupils too, so a pupil that looks aside leaves white behind.
      const eyeCells = EYE_ROWS.length * (EYE_WHITES.length + PUPILS.length);
      assert.equal(cells(whites).length, eyeCells, `${name}: whites`);
      assert.equal(cells(pupils).length, EYE_ROWS.length * PUPILS.length, `${name}: pupils`);

      assert.equal(whites, "", `${name}: no whites layer`);
      assert.equal(pupils, "", `${name}: no pupils layer`);
    }

  const skin = new Set(cells(spriteLayers(BASE).paths.find((p) => p.key === "S").d));

      assert.ok(skin.has(`${c + 1},${r + 1}`), `skin under eye pixel ${c},${r}`);
    }
  }

  assert.deepEqual(Object.keys(POSE_LAYERS).sort(), Object.keys(POSES).sort());

// Mini Justine, drawn as pixel maps: one character per pixel, "." empty.
// Spec: docs/superpowers/specs/2026-10-12-mascot-design.md. Pure (no DOM),
// so node --test covers it. The palette is character art with fixed colours;
// only the outline is themed, through --mascot-outline in globals.css.

// Every pose map is SIZE x SIZE; the drawn sprite is SIZE + 2 with the outline.
export const SIZE = 22;

export const PALETTE = {
  W: "#f4f4f2", // eye whites, envelope
  K: "#151517", // pupils, tee, shoes, sunglasses
  H: "#1e1b1a", // hair
  h: "#3d3634", // hair highlight
  S: "#e6b38c", // skin
  s: "#c48f6a", // mouth
  L: "#cfcfd3", // hoodie
  M: "#9a9aa1", // hood rim, laptop, envelope folds
  D: "#55555c", // trousers, laptop edge
  B: "#3b5bcc", // hoodie print
  G: "#3fbf73", // laptop logo, envelope seal
};

// The idle pose. The body is centred on column 9; the two spare columns on
// the right hold the raised arm when waving.
export const BASE = [
  "......HHHHHHH.........",
  "....HHHHHHHhHHH.......",
  "...HHHHHHHHHhHHH......",
  "..HHHHHHHHHHHHHHH.....",
  "..HHHHHHHHHHHHHHHH....",
  ".HHHHHHHHHHHHHHHHH....",
  ".HHHSHHHHSHHHHSHHH....",
  "..HSSSSSSSSSSSSSH.....",
  "..HSSSSSSSSSSSSSH.....",
  "..SSSWKWSSSWKWSSS.....",
  "..SSSWKWSSSWKWSSS.....",
  "..SSSSSSSSSSSSSSS.....",
  "...SSSSSSsSSSSSS......",
  "....SSSSSSSSSSS.......",
  "...MMMMKKKKKMMMM......",
  "..MLLLLKKKKKLLLLM.....",
  ".LLLLLLLLLLLLLLLLL....",
  ".LLLLBBBLLLBBBLLLL....",
  ".LLLLLLLLLLLLLLLLL....",
  ".SLLLLLLLLLLLLLLLS....",
  "...DDDDD...DDDDD......",
  "...KKKKK...KKKKK......",
];

// The eye pixels: rows 9-10, whites at columns 5, 7, 11 and 13, pupils at 6
// and 12. They are drawn as their own layers so the eyes can blink and the
// pupils can look left or right without redrawing the face.
export const EYE_ROWS = [9, 10];
export const EYE_WHITES = [5, 7, 11, 13];
export const PUPILS = [6, 12];

// A patch writes `text` at (row, col): "." clears a pixel, " " keeps it.
export const patch = (map, edits) => {
  const rows = map.map((row) => row.split(""));
  for (const [row, col, text] of edits) {
    [...text].forEach((ch, i) => {
      if (ch !== " ") rows[row][col + i] = ch;
    });
  }
  return rows.map((row) => row.join(""));
};

// Right arm raised (the viewer's right), hand beside the head.
const WAVE_A = [
  [8, 19, "SS"],
  [9, 19, "SS"],
  [10, 19, "L"],
  [11, 18, "LL"],
  [12, 18, "L"],
  [13, 17, "LL"],
  [14, 16, "LL"],
  [16, 17, "."],
  [17, 17, "."],
  [18, 17, "."],
  [19, 17, "."],
];
// The second wave frame tips the hand one pixel outward.
const WAVE_B = [
  [8, 19, ".SS"],
  [9, 19, ".SS"],
  [10, 19, "LL"],
];
// Two lenses joined by a bridge on the top row, a glint in each.
const SUNGLASSES = [
  [9, 4, "KKKKKKKKKKK"],
  [10, 4, "KKKK"],
  [10, 11, "KKKK"],
  [10, 8, "SSS"],
  [9, 5, "W"],
  [9, 12, "W"],
];
// A laptop held in front of the torso, its back to the viewer.
const LAPTOP = [
  [15, 3, "DDDDDDDDDDDDD"],
  [16, 3, "DMMMMMMMMMMMD"],
  [17, 3, "DMMMMMGMMMMMD"],
  [18, 3, "DMMMMMMMMMMMD"],
  [19, 2, "SDDDDDDDDDDDDDS"],
  [19, 1, "."],
  [19, 17, "."],
];
const ENVELOPE = [
  [15, 5, "WWWWWWWWW"],
  [16, 5, "WMWWWWWMW"],
  [17, 5, "WWMWWWMWW"],
  [18, 5, "WWWMGMWWW"],
  [19, 5, "WWWWWWWWW"],
  [17, 4, "S"],
  [17, 14, "S"],
  [19, 1, "."],
  [19, 17, "."],

export const POSES = {
  idle: BASE,
  waveA: patch(BASE, WAVE_A),
  waveB: patch(BASE, [...WAVE_A, ...WAVE_B]),
  sunglasses: patch(BASE, SUNGLASSES),
  typingA: patch(BASE, [...LAPTOP, [18, 2, "S"]]),
  typingB: patch(BASE, [...LAPTOP, [18, 16, "S"]]),
  envelope: patch(BASE, ENVELOPE),
};

// True when a pose shows the standard eyes (the sunglasses pose does not).
export const hasEyes = (map) =>
  EYE_ROWS.every(
    (r) => EYE_WHITES.every((c) => map[r][c] === "W") && PUPILS.every((c) => map[r][c] === "K"),
  );

// Pads the map by one pixel, then marks every empty pixel that touches a
// filled one (up, down, left or right) as outline "O".
export const withOutline = (map) => {
  const blank = ".".repeat(map[0].length + 2);
  const grid = [blank, ...map.map((row) => `.${row}.`), blank];
  const filled = (y, x) => {
    const ch = grid[y]?.[x];
    return ch !== undefined && ch !== ".";
  };
  return grid.map((row, y) =>
    [...row]
      .map((ch, x) => {
        if (ch !== ".") return ch;
        const touches = filled(y - 1, x) || filled(y + 1, x) || filled(y, x - 1) || filled(y, x + 1);
        return touches ? "O" : ".";
      })
      .join(""),
  );
};

// One SVG path for one colour: every horizontal run becomes a 1-high rect.
const runsPath = (rows, key) => {
  let d = "";
  rows.forEach((row, y) => {
    let x = 0;
    while (x < row.length) {
      if (row[x] !== key) {
        x += 1;
        continue;
      }
      let end = x;
      while (row[end] === key) end += 1;
      d += `M${x} ${y}h${end - x}v1h-${end - x}z`;
      x = end;
    }
  });
  return d;
};

// Single pixels at map (row, col) positions, shifted by the outline padding.
const cellsPath = (rows, cols) =>
  rows.flatMap((r) => cols.map((c) => `M${c + 1} ${r + 1}h1v1h-1z`)).join("");

// What a pose is drawn from: one path per colour (key "O" is the outline)
// and, for poses with eyes, separate whites and pupils paths. The whites
// cover all six eye columns, so a pupil shifted aside leaves white behind;
// the face under the eyes is filled with skin, so a blink shows skin.
export const spriteLayers = (map) => {
  const eyes = hasEyes(map);
  const eyeCols = [...EYE_WHITES, ...PUPILS];
  const body = withOutline(map).map((row, y) =>
    [...row]
      .map((ch, x) => (eyes && EYE_ROWS.includes(y - 1) && eyeCols.includes(x - 1) ? "S" : ch))
      .join(""),
  );
  const paths = [...Object.keys(PALETTE), "O"]
    .map((key) => ({ key, d: runsPath(body, key) }))
    .filter((layer) => layer.d);
  return {
    size: SIZE + 2,
    paths,
    whites: eyes ? cellsPath(EYE_ROWS, eyeCols) : "",
    pupils: eyes ? cellsPath(EYE_ROWS, PUPILS) : "",
  };
};

// Every pose's layers, computed once at import.
export const POSE_LAYERS = Object.fromEntries(
  Object.entries(POSES).map(([name, map]) => [name, spriteLayers(map)]),
);

- [ ] **Step 4: Write `src/component/mascot/HeroMascot.jsx`**

```jsx
"use client";

import { useEffect, useRef, useState, useSyncExternalStore } from "react";
import Sprite from "@/component/mascot/Sprite";
import {
  CLICK_LINES,
  FALLBACK_GREETING,
  THEME_LINES,
  greeting,
  nextLineIndex,
} from "@/lib/mascot/lines";

// The hop on click: squash, jump, settle (about half a second).
const HOP = [
  { transform: "translateY(0) scale(1, 1)" },
  { transform: "translateY(0) scale(1.12, 0.86)", offset: 0.15 },
  { transform: "translateY(-14px) scale(0.94, 1.08)", offset: 0.45 },
  { transform: "translateY(0) scale(1.08, 0.92)", offset: 0.75 },
  { transform: "translateY(0) scale(1, 1)" },
];
const SUNGLASSES_MS = 4000;
// The pupils look left or right only once the pointer is this far from the
// mascot's centre, so they rest while the pointer is on it.
const LOOK_DEADZONE_PX = 24;

const reducedMotion = () => matchMedia("(prefers-reduced-motion: reduce)").matches;

const hop = (el) => {
  if (!el || reducedMotion()) return;
  el.animate(HOP, { duration: 520, easing: "cubic-bezier(0.3, 0.7, 0.4, 1)" });
};

// The visitor's hour, read on the client only. The server snapshot is null,
// so the server HTML and the first client render show the fallback greeting
// and hydration always matches.
const noSubscribe = () => () => {};
const getHour = () => new Date().getHours();
const getServerHour = () => null;

// Mini Justine in the hero: waves on load (CSS), greets by the visitor's
// time, follows a fine pointer with its eyes, hops and says the next line
// when clicked, and reacts to the theme toggle. Spec:
// docs/superpowers/specs/2026-10-12-mascot-design.md.
const HeroMascot = () => {
  const hour = useSyncExternalStore(noSubscribe, getHour, getServerHour);
  const [lineIndex, setLineIndex] = useState(-1); // -1 until the first click
  const [themeLine, setThemeLine] = useState(null);
  const [pose, setPose] = useState("idle"); // "idle" or "sunglasses"
  const buttonRef = useRef(null);
  const hopRef = useRef(null);

  // Eyes follow a fine pointer: --look on the button, no React state.
  useEffect(() => {
    if (!matchMedia("(pointer: fine)").matches || reducedMotion()) return undefined;
    let frame = 0;
    const onMove = (e) => {
      if (frame) return;
      frame = requestAnimationFrame(() => {
        frame = 0;
        const el = buttonRef.current;
        if (!el) return;
        const rect = el.getBoundingClientRect();
        const dx = e.clientX - (rect.left + rect.width / 2);
        let look = 0;
        if (dx > LOOK_DEADZONE_PX) look = 1;
        if (dx < -LOOK_DEADZONE_PX) look = -1;
        el.style.setProperty("--look", String(look));
      });
    };
    window.addEventListener("pointermove", onMove);
    return () => {
      window.removeEventListener("pointermove", onMove);
      if (frame) cancelAnimationFrame(frame);
    };
  }, []);

  // React to the theme toggle (data-theme on <html>), not to an OS change.
  useEffect(() => {
    let timer;
    const observer = new MutationObserver(() => {
      const theme = document.documentElement.dataset.theme;
      if (theme !== "light" && theme !== "dark") return;
      clearTimeout(timer);
      setThemeLine(THEME_LINES[theme]);
      if (theme === "light") {
        setPose("sunglasses");
        timer = setTimeout(() => setPose("idle"), SUNGLASSES_MS);
      } else {
        setPose("idle");
        hop(hopRef.current);
      }
    });
    observer.observe(document.documentElement, {
      attributes: true,
      attributeFilter: ["data-theme"],
    });
    return () => {
      observer.disconnect();
      clearTimeout(timer);
    };
  }, []);

  const onClick = () => {
    setThemeLine(null);
    setLineIndex(nextLineIndex);
    hop(hopRef.current);
  };

  let line = FALLBACK_GREETING;
  if (themeLine) line = themeLine;
  else if (lineIndex >= 0) line = CLICK_LINES[lineIndex];
  else if (hour !== null) line = greeting(hour);

  return (
    <div className="rise-in mt-8 flex items-center gap-4" style={{ "--i": 3 }}>
      <button
        ref={buttonRef}
        type="button"
        onClick={onClick}
        aria-label="Mini Justine, say something"
        data-pose={pose}
        className="mascot relative size-24 shrink-0"
      >
        <span ref={hopRef} className="block size-full origin-bottom">
          <span className="mascot-bob relative block size-full">
            <span className="mascot-layer mascot-wave">
              <Sprite pose="waveA" />
              <Sprite pose="waveB" className="mascot-frame-b" />
            </span>
            <span className="mascot-layer mascot-idle">
              <Sprite pose="idle" />
            </span>
            <span className="mascot-layer mascot-sunglasses">
              <Sprite pose="sunglasses" />
            </span>
          </span>
        </span>
      </button>
      <p
        aria-live="polite"
        className="mascot-bubble rounded-card border border-line-strong bg-surface px-3 py-1.5 text-sm text-ink"
      >
        {line}
      </p>
    </div>
  );
};

export default HeroMascot;
```

"use client";

import { useState } from "react";
import Sprite from "@/component/mascot/Sprite";
import { CAMEO_LINES } from "@/lib/mascot/lines";

// Pose frames per cameo: typing alternates two frames, the envelope bobs.
const FRAMES = {
  typing: ["typingA", "typingB"],
  envelope: ["envelope"],
};

// Mini Justine standing on a section heading's hairline (pass it as the
// heading's `action`): typing in Tech stack, holding an envelope in Contact.
// Its line shows on hover or keyboard focus (CSS) and on tap (data-open),
// since touch has no hover.
const Cameo = ({ kind }) => {
  const [open, setOpen] = useState(false);
  const [first, second] = FRAMES[kind];
  const line = CAMEO_LINES[kind];
  const motion = second ? "mascot-typing" : "mascot-bob";

  return (
    <span className="mascot-cameo relative -mb-3 inline-flex" data-open={open}>
      <span
        aria-hidden="true"
        className="mascot-cameo-line absolute bottom-full right-0 mb-1 whitespace-nowrap rounded-card border border-line-strong bg-surface px-2.5 py-1 text-xs text-ink"
      >
        {line}
      </span>
      <button
        type="button"
        onClick={() => setOpen((value) => !value)}
        onBlur={() => setOpen(false)}
        aria-label={`Mini Justine: ${line}`}
        className={`${motion} relative size-12`}
      >
        <Sprite pose={first} />
        {second && <Sprite pose={second} className="mascot-frame-b" />}
      </button>
    </span>
  );
};

export default Cameo;
